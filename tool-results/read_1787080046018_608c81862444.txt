   131→  async adaptTone(
   132→    content: string,
   133→    targetAudience: "academic" | "business" | "government"
   134→  ): Promise<string> {
   135→    const toneGuides = {
   136→      academic: {
   137→        style: "formal academic writing with citations and technical depth",
   138→        vocabulary: "specialized terminology appropriate for peer review",
   139→        structure: "structured with clear methodology and evidence-based claims",
   140→      },
   141→      business: {
   142→        style: "professional business language focused on value proposition",
   143→        vocabulary: "business terminology emphasizing ROI and market opportunity",
   144→        structure: "executive summary format with clear value drivers",
   145→      },
   146→      government: {
   147→        style: "clear policy-oriented language aligned with public benefit",
   148→        vocabulary: "terms aligned with government priorities and frameworks",
   149→        structure: "aligned with evaluation criteria and measurable outcomes",
   150→      },
   151→    };
   152→
   153→    const guide = toneGuides[targetAudience];
   154→    const prompt = `Adapt the following text for a ${targetAudience} audience.
   155→
   156→Target Style: ${guide.style}
   157→Vocabulary: ${guide.vocabulary}
   158→Structure: ${guide.structure}
   159→
   160→Original Text:
   161→${content}
   162→
   163→Adapted Text:`;
   164→
   165→    if (this.provider.name === "mock") {
   166→      return this.mockToneAdaptation(content, targetAudience);
   167→    }
   168→
   169→    try {
   170→      const result = await this.callLLM(prompt, 3000);
   171→      return result;
   172→    } catch (error) {
   173→      console.error("Error adapting tone:", error);
   174→      return content;
   175→    }
   176→  }

   178→  /**
   179→   * Check reading level of text
   180→   */
   181→  checkReadingLevel(text: string, _targetGrade: number): ReadingScore {
   182→    const words = text.split(/\s+/).filter(w => w.length > 0);
   183→    const sentences = text.split(/[.!?]+/).filter(s => s.trim().length > 0);
   184→    const syllables = words.reduce((sum, word) => sum + this.countSyllables(word), 0);
   185→
   186→    const wordCount = Math.max(words.length, 1);
   187→    const sentenceCount = Math.max(sentences.length, 1);
   188→
   189→    // Flesch-Kincaid Grade Level
   190→    const fleschKincaid = (
   191→      0.39 * (wordCount / sentenceCount) +
   192→      11.8 * (syllables / wordCount) -
   193→      15.59
   194→    );
   195→
   196→    // Coleman-Liau Index
   197→    const avgWordLength = wordCount > 0 ? words.reduce((sum, w) => sum + w.length, 0) / wordCount : 0;
   198→    const colemanLiau = (0.0588 * avgWordLength * 100) - (0.296 * (sentenceCount / wordCount) * 100) - 15.8;
   199→
   200→    return {
   201→      gradeLevel: Math.round(fleschKincaid),
   202→      fleschKincaid: Math.round(fleschKincaid * 10) / 10,
   203→      colemanLiau: Math.round(colemanLiau * 10) / 10,
   204→      wordsPerSentence: Math.round((wordCount / sentenceCount) * 10) / 10,
   205→      syllablesPerWord: Math.round((syllables / wordCount) * 100) / 100,
   206→    };
   207→  }

   209→  /**
   210→   * Check similarity to reference texts (for plagiarism/originality)
   211→   */
   212→  checkSimilarityToWinners(text: string, referenceTexts: string[]): SimilarityScore {
   213→    const normalizedText = text.toLowerCase().replace(/[^\w\s]/g, "");
   214→    const textWords = new Set(normalizedText.split(/\s+/));
   215→
   216→    const matches: SimilarityScore["matches"] = [];
   217→    let totalSimilarity = 0;
   218→
   219→    for (const ref of referenceTexts) {
   220→      const normalizedRef = ref.toLowerCase().replace(/[^\w\s]/g, "");
   221→      const refWords = normalizedRef.split(/\s+/);
   222→
   223→      // Simple Jaccard similarity
   224→      const commonWords = [...textWords].filter(w => refWords.includes(w));
   225→      const allWords = new Set([...textWords, ...refWords]);
   226→      const similarity = commonWords.length / allWords.size;
   227→
   228→      if (similarity > 0.3) {
   229→        matches.push({
   230→          text: commonWords.slice(0, 5).join(" "),
   231→          reference: ref.substring(0, 100),
   232→          similarity: Math.round(similarity * 100) / 100,
   233→        });
   234→        totalSimilarity = Math.max(totalSimilarity, similarity);
   235→      }
   236→    }
   237→
   238→    return {
   239→      percentage: Math.round(totalSimilarity * 100),
   240→      matches: matches.slice(0, 5),
   241→      overallAssessment: totalSimilarity > 0.5 
   242→        ? "High similarity detected - consider rephrasing"
   243→        : totalSimilarity > 0.3
   244→        ? "Moderate similarity - acceptable but could be more original"
   245→        : "Good originality - low similarity to references",
   246→    };
   247→  }

   251→  private async callLLM(prompt: string, maxTokens: number): Promise<string> {
   252→    if (this.provider.name === "openai" && this.provider.apiKey) {
   253→      return await this.callOpenAI(prompt, maxTokens);
   254→    } else if (this.provider.name === "anthropic" && this.provider.apiKey) {
   255→      return await this.callAnthropic(prompt, maxTokens);
   256→    } else {
   257→      return this.generateMockResponse(prompt);
   258→    }
   259→  }
   260→
   261→  private async callOpenAI(prompt: string, maxTokens: number): Promise<string> {
   262→    const response = await fetch("https://api.openai.com/v1/chat/completions", {
   263→      method: "POST",
   264→      headers: {
   265→        "Content-Type": "application/json",
   266→        Authorization: `Bearer ${this.provider.apiKey}`,
   267→      },
   268→      body: JSON.stringify({
   269→        model: this.provider.model,
   270→        messages: [
   271→          {
   272→            role: "system",
   273→            content: "You are an expert grant writer specializing in research funding applications. Write clear, compelling, and professional content that maximizes success rates.",
   274→          },
   275→          { role: "user", content: prompt },
   276→        ],
   277→        max_tokens: maxTokens,
   278→        temperature: 0.7,
   279→      }),
   280→    });
   281→
   282→    if (!response.ok) {
   283→      throw new Error(`OpenAI API error: ${response.status}`);
   284→    }
   285→
   286→    const data = await response.json();
   287→    return data.choices[0].message.content;
   288→  }

   290→  private async callAnthropic(prompt: string, maxTokens: number): Promise<string> {
   291→    const response = await fetch("https://api.anthropic.com/v1/messages", {
   292→      method: "POST",
   293→      headers: {
   294→        "Content-Type": "application/json",
   295→        "x-api-key": this.provider.apiKey!,
   296→        "anthropic-version": "2023-06-01",
   297→      },
   298→      body: JSON.stringify({
   299→        model: this.provider.model,
   300→        max_tokens: maxTokens,
   301→        messages: [
   302→          {
   303→            role: "user",
   304→            content: `You are an expert grant writer specializing in research funding applications. Write clear, compelling, and professional content.\n\n${prompt}`,
   305→          },
   306→        ],
   307→      }),
   308→    });
   309→
   310→    if (!response.ok) {
   311→      throw new Error(`Anthropic API error: ${response.status}`);
   312→    }
   313→
   314→    const data = await response.json();
   315→    return data.content[0].text;
   316→  }

   318→  private async generateContent(
   319→    prompt: string,
   320→    maxWords: number,
   321→    fieldType: SemanticFieldType
   322→  ): Promise<GeneratedContent> {
   323→    try {
   324→      const content = await this.callLLM(prompt, maxWords * 2); // Approximate tokens
   325→      
   326→      const wordCount = content.split(/\s+/).length;
   327→      const readingTime = Math.ceil(wordCount / 200); // Average reading speed
   328→
   329→      return {
   330→        content,
   331→        wordCount,
   332→        readingTime,
   333→        confidence: this.provider.name === "mock" ? 0.7 : 0.9,
   334→        suggestions: this.generateSuggestions(fieldType),
   335→      };
   336→    } catch (error) {
   337→      console.error(`Error generating ${fieldType}:`, error);
   338→      
   339→      // Return mock content as fallback
   340→      const fallbackContent = this.getMockContent(fieldType);
   341→      return {
   342→        content: fallbackContent,
   343→        wordCount: fallbackContent.split(/\s+/).length,
   344→        readingTime: Math.ceil(fallbackContent.split(/\s+/).length / 200),
   345→        confidence: 0.5,
   346→        suggestions: ["Consider providing custom content for best results"],
   347→      };
   348→    }
   349→  }

   918→  private countSyllables(word: string): number {
   919→    word = word.toLowerCase().replace(/[^a-z]/g, "");
   920→    if (word.length <= 3) return 1;
   921→    
   922→    let count = 0;
   923→    const vowels = "aeiouy";
   924→    let prevIsVowel = false;
   925→
   926→    for (const char of word) {
   927→      const isVowel = vowels.includes(char);
   928→      if (isVowel && !prevIsVowel) count++;
   929→      prevIsVowel = isVowel;
   930→    }
   931→
   932→    return Math.max(count, 1);
   933→  }
